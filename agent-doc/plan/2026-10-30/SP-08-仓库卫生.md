**verify_cmd**

```powershell
pwsh -NoProfile -Command "if(-not (Test-Path .gitignore)){exit 1}; if(Test-Path 'plan'){exit 1}; $bad = Get-ChildItem -Recurse -Directory tools | Where-Object { $_.Name -like '*- Copy*' }; if($bad){exit 1}; $zips = Get-ChildItem -Recurse -File tools -Filter *.zip; if($zips){exit 1}; exit 0"
```
